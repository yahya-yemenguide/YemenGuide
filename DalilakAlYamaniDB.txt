CREATE TABLE dbo.Users
(
    UserID INT IDENTITY(1,1) NOT NULL,
    FullName NVARCHAR(100) NOT NULL,
    Username NVARCHAR(50) NOT NULL,
    PasswordHash NVARCHAR(255) NOT NULL,
    ProfileImage NVARCHAR(255) NULL,
    ContactInfo NVARCHAR(100) NULL,
    PhoneNumber NVARCHAR(50) NULL,
    CreatedAt DATETIME NULL CONSTRAINT DF_Users_CreatedAt DEFAULT GETDATE(),
    Role NVARCHAR(20) NOT NULL,
    AccountState NVARCHAR(20) NULL CONSTRAINT DF_Users_AccountState DEFAULT N'Active',
    IsDarkMode BIT NOT NULL CONSTRAINT DF_Users_IsDarkMode DEFAULT 0,
    CommercialRegisterDocument NVARCHAR(255) NULL,
    IdentityDocument NVARCHAR(255) NULL,
    AuthorizationDocument NVARCHAR(255) NULL,
    IsOnline BIT NOT NULL CONSTRAINT DF_Users_IsOnline DEFAULT 0,
    LastSeen DATETIME NULL,
    WarningCount INT NOT NULL CONSTRAINT DF_Users_WarningCount DEFAULT 0,
    LastWarningDate DATETIME NULL,
    IsEmailVerified BIT NOT NULL CONSTRAINT DF_Users_IsEmailVerified DEFAULT 0,
    VerificationCode NVARCHAR(10) NULL,
    CodeExpiry DATETIME NULL,
    Bio NVARCHAR(MAX) NULL,
    DateOfBirth DATE NULL,
    Gender NVARCHAR(10) NULL,
    City NVARCHAR(100) NULL,
    WalletBalance DECIMAL(18,2) NOT NULL CONSTRAINT DF_Users_WalletBalance DEFAULT 0.00,
    FacebookUrl NVARCHAR(255) NULL,
    TwitterUrl NVARCHAR(255) NULL,
    InstagramUrl NVARCHAR(255) NULL,
    PreferredCurrency NVARCHAR(50) NOT NULL CONSTRAINT DF_Users_PreferredCurrency DEFAULT N'ريال يمني (قديم)',
    ReceiveOffers BIT NOT NULL CONSTRAINT DF_Users_ReceiveOffers DEFAULT 1,
    JobTitle NVARCHAR(100) NULL,
    IsAgreedToTerms BIT NOT NULL CONSTRAINT DF_Users_IsAgreedToTerms DEFAULT 0,
    AgreedToTermsDate DATETIME NULL,
    PreferredLanguage NVARCHAR(10) NOT NULL CONSTRAINT DF_Users_PreferredLanguage DEFAULT N'ar',

    FreeTrialStartedAt DATETIME2 NULL,
    FreeTrialEndsAt DATETIME2 NULL,

    AdminRoleID INT NULL,
    EmployeeNumber NVARCHAR(30) NULL,
    Department NVARCHAR(100) NULL,
    EmploymentStatus NVARCHAR(20) NULL,
    HireDate DATE NULL,
    EndDate DATE NULL,
    LastLoginAt DATETIME NULL,
    MustChangePassword BIT NOT NULL CONSTRAINT DF_Users_MustChangePassword DEFAULT 0,

    CONSTRAINT PK_Users
        PRIMARY KEY CLUSTERED (UserID),

    CONSTRAINT CK_Users_EmploymentStatus
        CHECK
        (
            EmploymentStatus IS NULL
            OR EmploymentStatus IN
            (
                N'Active',
                N'Suspended',
                N'OnLeave',
                N'Terminated'
            )
        )
);
GO

        CONSTRAINT DF_Categories_ThemeColor DEFAULT N'#0d6efd',
    ImagePath NVARCHAR(MAX) NULL,
    BookingSystemId INT NOT NULL
        CONSTRAINT DF_Categories_BookingSystemId DEFAULT 0,
    CommissionPercentage DECIMAL(5,2) NULL,

    -- 1 = المالك والإدارة يستطيعان إضافة منشآت

    CONSTRAINT PK_Categories
        PRIMARY KEY CLUSTERED (CategoryID),

    CONSTRAINT CK_Categories_CommissionPercentage
        CHECK
        (
            CommissionPercentage IS NULL
            OR
            (
                CommissionPercentage >= 0
                AND CommissionPercentage <= 100
            )
        )
);

        ),

    CONSTRAINT CK_Advertisements_CompleteStatistics
